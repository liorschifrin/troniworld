} function handleMentionInput(e) { const inputEl = e.currentTarget; updatePlaceholder(inputEl); // Defer by one tick so iOS Safari has updated the selection before we read it setTimeout(() => { const q = getMentionQuery(inputEl); } } function checkWordAtCaret(inputEl) { const sel = window.getSelection(); if (!sel || sel.rangeCount === 0) return; const range = sel.getRangeAt(0); if (!inputEl.contains(range.startContainer) || !range.collapsed) return; } function insertMentionAt(inputEl) { // Don't call focus() — onmousedown/ontouchstart + preventDefault keeps focus in the input. // Calling focus() triggers a repaint which causes visible delay. const sel = window.getSelection();
